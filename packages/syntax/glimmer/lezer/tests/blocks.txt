# If else

{{#if a}}yes{{else}}no{{/if}}

==>

Document(
  IfBlock(
    IfBlockOpen(
      "{{",
      BlockPrefix,
      BlockType,
      LongExpression,
      "}}"
    ),
    Text,
    ElseBlock(
      "{{",
      BlockType,
      "}}"
    ),
    Text,
    IfBlockClose(
      "{{",
      BlockPrefix,
      BlockType,
      "}}"
    )
  )
)
